cmake_minimum_required(VERSION 3.15)
project(asm6x CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
file(GLOB SOURCES src/*.cpp)
add_executable(asm6x ${SOURCES})
if(MSVC)
  target_compile_options(asm6x PRIVATE /W4 /permissive-)
else()
  target_compile_options(asm6x PRIVATE -Wall -Wextra -pedantic)
  find_package(Threads REQUIRED)
  target_link_libraries(asm6x Threads::Threads)
endif()
